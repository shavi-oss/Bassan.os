import { Entity, Column, PrimaryColumn, CreateDateColumn, UpdateDateColumn } from 'typeorm';

@Entity('companies')
export class Company {
  @PrimaryColumn('varchar', { length: 50 })
  id: string;

  @Column('varchar', { length: 255 })
  name: string;

  @Column('char', { length: 2 })
  countryCode: string;

  @Column('char', { length: 3 })
  currency: string;

  @Column('jsonb')
  settings: Record<string, any>;

  @Column('varchar', { length: 20 })
  status: string;

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

// backend/src/modules/company/company.service.ts
@Injectable()
export class CompanyService {
  async createCompany(data: CreateCompanyDto): Promise<Company> {
    const company = this.companyRepository.create({
      id: `COMP_${uuidv4().replace(/-/g, '_')}`,
      ...data
    });
    return this.companyRepository.save(company);
  }
}

// backend/src/modules/commission/commission.calculator.ts
export class CommissionCalculator {
  async calculateCommission(sale: Sale, employee: Employee): Promise<Commission> {
    // 1. احصل على القواعد المناسبة
    const rules = await this.getApplicableRules(employee, sale);
    
    // 2. طبّق القواعد
    const amount = this.applyRules(rules, sale.amount);
    
    // 3. تحقق من الميزانيات
    await this.checkBudgets(employee, amount);
    
    // 4. أنشئ سجل العمولة
    return this.createCommissionRecord(sale, employee, amount, rules);
  }
}

// backend/src/modules/integration/shopify/shopify.service.ts
@Injectable()
export class ShopifyService {
  async handleWebhook(event: string, data: any): Promise<void> {
    // 1. تحقق من التوقيع
    this.verifySignature(event, data);
    
    // 2. حوّل البيانات إلى نموذج Sale
    const sale = this.transformToSale(data);
    
    // 3. أطلق حدث NEW_SALE
    await this.eventEmitter.emit('SALE_CREATED', {
      sale,
      source: 'shopify'
    });
  }
}
2. تكامل Telegram
typescript
// telegram-bot/src/bot.service.ts
export class TelegramBotService {
  @Command('start')
  async startCommand(ctx: Context): Promise<void> {
    const user = await this.authService.authenticateTelegramUser(ctx.from);
    await ctx.reply(`مرحباً ${user.fullName}!`, {
      reply_markup: this.getMainMenuKeyboard()
    });
  }

  @Command('commissions')
  async commissionsCommand(ctx: Context): Promise<void> {
    const commissions = await this.commissionService.getEmployeeCommissions(
      ctx.from.id
    );
    
    const message = this.formatCommissionsMessage(commissions);
    await ctx.reply(message, { parse_mode: 'HTML' });
  }
}

version: '3.8'

services:
  postgres:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: bassan_prod
      POSTGRES_USER: bassan_app
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./init-scripts:/docker-entrypoint-initdb.d
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U bassan_app"]
      interval: 10s
      timeout: 5s
      retries: 5

  redis:
    image: redis:7-alpine
    command: redis-server --requirepass ${REDIS_PASSWORD}
    volumes:
      - redis_data:/data
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 3

  backend:
    build:
      context: ../backend
      dockerfile: Dockerfile.prod
    environment:
      NODE_ENV: production
      DATABASE_URL: postgresql://bassan_app:${DB_PASSWORD}@postgres:5432/bassan_prod
      REDIS_URL: redis://:${REDIS_PASSWORD}@redis:6379
    ports:
      - "3000:3000"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy

  frontend:
    build:
      context: ../frontend
      dockerfile: Dockerfile.prod
    ports:

    depends_on:
