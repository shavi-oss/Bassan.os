3. التحقق من الصلاحيات في كل طلب:
typescript
// مثال للتحقق في Middleware
function authorize(requiredPermission: string, requiredScope?: string) {
  return (req, res, next) => {
    const user = req.user;
    
    // 1. التحقق من الصلاحية
    if (!user.permissions.includes(requiredPermission)) {
      return res.status(403).json({ error: 'غير مصرح' });
    }
    
    // 2. التحقق من النطاق
    if (requiredScope && user.scope !== requiredScope) {
      return res.status(403).json({ error: 'خارج النطاق المسموح' });
    }
    
    next();
  };
}

// استخدام في Route
app.get('/api/v1/commissions', 
  authorize('commissions:read', 'sales'),
  getCommissions
);

🗄️ تصميم قاعدة البيانات (Database Design)
1. مخطط العلاقات الأساسية:
sql
-- جدول الشركات
CREATE TABLE companies (
    id VARCHAR(50) PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    country_code CHAR(2) NOT NULL,
    currency CHAR(3) NOT NULL,
    plan VARCHAR(50) NOT NULL,
    settings JSONB DEFAULT '{}',
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- جدول الموظفين
CREATE TABLE employees (
    id VARCHAR(50) PRIMARY KEY,
    company_id VARCHAR(50) REFERENCES companies(id),
    user_id VARCHAR(50) UNIQUE,
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(20),
    department VARCHAR(50) NOT NULL,
    employment_type VARCHAR(20) NOT NULL,
    contract JSONB NOT NULL,
    status VARCHAR(20) DEFAULT 'active',
    permissions JSONB DEFAULT '[]',
    created_at TIMESTAMP DEFAULT NOW(),
    INDEX idx_company_department (company_id, department)
);

-- جدول العمولات
CREATE TABLE commissions (
    id VARCHAR(50) PRIMARY KEY,
    employee_id VARCHAR(50) REFERENCES employees(id),
    company_id VARCHAR(50) REFERENCES companies(id),
    amount DECIMAL(15,2) NOT NULL,
    currency CHAR(3) NOT NULL,
    calculation_details JSONB NOT NULL,
    status VARCHAR(20) NOT NULL,
    payment_status VARCHAR(20) NOT NULL,
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMP DEFAULT NOW(),
    paid_at TIMESTAMP,
    INDEX idx_employee_status (employee_id, status),
    INDEX idx_company_created (company_id, created_at)
);

-- جدول المهام
CREATE TABLE tasks (
    id VARCHAR(50) PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    assigned_to JSONB NOT NULL, -- Array of employee IDs
    department VARCHAR(50) NOT NULL,
    priority VARCHAR(20) NOT NULL,
    status VARCHAR(20) NOT NULL,
    due_date TIMESTAMP NOT NULL,
    dependencies JSONB DEFAULT '[]',
    attachments JSONB DEFAULT '[]',
    created_by VARCHAR(50) NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    completed_at TIMESTAMP,
    INDEX idx_department_status (department, status),
    INDEX idx_assigned_due (assigned_to, due_date)
);
2. استراتيجية التجزئة (Partitioning Strategy):
sql
-- تجزئة جدول العمولات حسب الشركة والشهر
CREATE TABLE commissions_2024_05 (
    CHECK (company_id = 'COMP001' AND created_at >= '2024-05-01' AND created_at < '2024-06-01')
) INHERITS (commissions);

-- إنشاء الفهرس على الجدول المجزأ
CREATE INDEX idx_commissions_2024_05_employee 
ON commissions_2024_05 (employee_id, created_at);

-- توجيه البيانات تلقائياً
CREATE OR REPLACE FUNCTION commissions_insert_trigger()
RETURNS TRIGGER AS $$
BEGIN
    IF (NEW.company_id = 'COMP001' AND NEW.created_at >= '2024-05-01' AND NEW.created_at < '2024-06-01') THEN
        INSERT INTO commissions_2024_05 VALUES (NEW.*);
    ELSE
        -- Handle other partitions
    END IF;
    RETURN NULL;
END;
$$
LANGUAGE plpgsql;
3. سياسات الاستعلام الأمثل:
sql
-- استعلام فعال لجلب عمولات موظف لشهر معين
SELECT 
    c.*,
    e.full_name,
    e.department
FROM commissions c
JOIN employees e ON c.employee_id = e.id
WHERE c.company_id = 'COMP001'
    AND c.employee_id = 'EMP001'
    AND c.created_at >= '2024-05-01'
    AND c.created_at < '2024-06-01'
    AND c.status = 'approved'
ORDER BY c.created_at DESC
LIMIT 100;

-- استعلام تجميعي لأداء الأقسام
SELECT 
    e.department,
    COUNT(c.id) as commission_count,
    SUM(c.amount) as total_commissions,
    AVG(c.amount) as average_commission
FROM commissions c
JOIN employees e ON c.employee_id = e.id
WHERE c.company_id = 'COMP001'
    AND c.created_at >= '2024-05-01'
    AND c.created_at < '2024-06-01'
    AND c.status = 'paid'
GROUP BY e.department
ORDER BY total_commissions DESC;
